### Brakujący element: tabela `reminders`

```sql
-- Brak w migration 001 — dodać przed Sprint 3
CREATE TABLE reminders (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id     UUID NOT NULL REFERENCES profiles ON DELETE CASCADE,
  contact_id  UUID REFERENCES contacts ON DELETE CASCADE,  -- null = globalne przypomnienie
  title       TEXT NOT NULL,
  remind_at   TIMESTAMPTZ NOT NULL,
  is_done     BOOLEAN DEFAULT FALSE,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE reminders ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Own reminders" ON reminders
  FOR ALL USING (auth.uid() = user_id);
-- Index dla zapytań po dacie
CREATE INDEX idx_reminders_remind_at ON reminders(user_id, remind_at) WHERE is_done = FALSE;
```
